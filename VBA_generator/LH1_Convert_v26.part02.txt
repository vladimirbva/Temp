'=====================================================================
' Ф32-пост-резка vv25 (L9) - перенос механики Ф32а/Ф32б из LH3 v24 на
' уровень ГОТОВОГО файла. Цель: исключить FPC "Procedure too complex"
' (регистровая сложность) на ЛЮБОЙ процедуре выхлопа. Работает ПОСЛЕ
' записи выхлопа, основную логику конвертора не меняет.
'   пороги: часть > 5000 строк -> резать; монолит-блок или тело
'   цикла-переростка > 4000 -> Ф32а (в-блочная резка с дублированием
'   условия) / Ф32б (вынос тела цикла в части-процедуры, вызовы внутри
'   цикла). Если резка невозможна - остаётся как есть. Повторная обработка
'   файла защищена маркером "Ф32-пост-резка v25".
'=====================================================================
' LH1 v26 Ф4: модульные Const L9_THR_* УДАЛЕНЫ - у владельца они терялись при переносе вставки
' ("L9_THR_LINE - variable not defined"). Теперь константы объявлены ЛОКАЛЬНО внутри
' каждой использующей процедуры (L9_CutOne, L9_CutFile) - вставка самодостаточна.
' ПРАВИЛО (25.11): перед выдачей КАЖДОЙ версии - прогон vba_decl_check.py (необъявленные
' идентификаторы + зависимость вставок от внешних объявлений). Без прогона версию не выдавать.

Private Function L9_Delta(ByVal s As String) As Long
    Dim q As Boolean, i As Long, c As String, buf As String, t As String
    Dim d As Long, j As Long, w As String
    q = False: buf = ""
    For i = 1 To Len(s)
        c = Mid(s, i, 1)
        If c = "'" Then
            q = Not q: buf = buf & c
        ElseIf c = "/" And (Not q) And i < Len(s) And Mid(s, i + 1, 1) = "/" Then
            Exit For
        Else
            buf = buf & c
        End If
    Next i
    t = LCase(buf)
    If InStr(t, "begin") = 0 And InStr(t, "end") = 0 And InStr(t, "try") = 0 _
       And InStr(t, "repeat") = 0 And InStr(t, "until") = 0 And InStr(t, "case") = 0 Then
        L9_Delta = 0: Exit Function
    End If
    For i = 1 To Len(t)
        c = Mid(t, i, 1)
        If Not ((c >= "a" And c <= "z") Or (c >= "0" And c <= "9") Or c = "_") Then Mid(t, i, 1) = " "
    Next i
    d = 0
    Dim parts() As String
    parts = Split(t, " ")
    For j = LBound(parts) To UBound(parts)
        w = parts(j)
        If w = "begin" Or w = "try" Or w = "repeat" Then d = d + 1
        If w = "end" Or w = "until" Then d = d - 1
        If w = "case" Then d = d + 1
    Next j
    L9_Delta = d
End Function

Private Function L9_Simple(ByVal s As String) As Boolean
    Dim i As Long, c As String
    If Len(s) = 0 Then L9_Simple = False: Exit Function
    If Not ((Left(s, 1) >= "a" And Left(s, 1) <= "z") Or (Left(s, 1) >= "A" And Left(s, 1) <= "Z") Or Left(s, 1) = "_") Then L9_Simple = False: Exit Function
    For i = 1 To Len(s)
        c = Mid(s, i, 1)
        If Not ((c >= "a" And c <= "z") Or (c >= "A" And c <= "Z") Or (c >= "0" And c <= "9") Or c = "_") Then L9_Simple = False: Exit Function
    Next i
    L9_Simple = True
End Function

Private Function L9_Suf(ByVal k As Long) As String
    Dim bs As Long, nn As Long, s As String
    bs = 0: nn = k
    Do While nn > 26
        nn = nn - 26
        bs = bs + 1
    Loop
    If bs > 0 Then s = Chr(96 + bs)
    L9_Suf = s & Chr(96 + nn)
End Function

Private Function L9_MonoStrip(ByVal s As String) As String
    Dim i As Long, q2 As Boolean, r As String, c As String
    q2 = False: r = ""
    For i = 1 To Len(s)
        c = Mid(s, i, 1)
        If c = "'" Then
            q2 = Not q2
        ElseIf c = "/" And (Not q2) And i < Len(s) And Mid(s, i + 1, 1) = "/" Then
            Exit For
        ElseIf Not q2 Then
            r = r & c
        End If
    Next i
    L9_MonoStrip = r
End Function

Private Function L9_MonoWords(ByVal s As String) As String()
    Dim r() As String, i As Long, c As String, buf As String, n As Long
    For i = 1 To Len(s)
        c = Mid(s, i, 1)
        If (c >= "a" And c <= "z") Or (c >= "0" And c <= "9") Or c = "_" Then
            buf = buf & c
        Else
            If Len(buf) > 0 Then
                n = n + 1
                ReDim Preserve r(1 To n)
                r(n) = buf
                buf = ""
            End If
        End If
    Next i
    If Len(buf) > 0 Then
        n = n + 1
        ReDim Preserve r(1 To n)
        r(n) = buf
    End If
    If n = 0 Then
        ReDim r(1 To 1): r(1) = ""
    End If
    L9_MonoWords = r
End Function

Private Sub L9_CutCand(ByRef body() As String, ByVal bn As Long, ByRef cand() As Long, ByRef ncand As Long, ByVal depth As Long)
    Dim k As Long, dd As Long, t As String, jn As Long, nx As String, badn As Boolean
    ncand = 0: dd = 0
    For k = 1 To bn
        dd = dd + L9_Delta(body(k))
        If dd = depth Then
            t = Trim(body(k))
            If Left(t, 2) = "//" Then
                ncand = ncand + 1
                If ncand > UBound(cand) Then ReDim Preserve cand(1 To ncand * 2)
                cand(ncand) = k
            ElseIf Right(t, 1) = ";" And LCase(Left(t, 3)) <> "end" And LCase(Left(t, 4)) <> "else" Then
                jn = k + 1
                Do While jn <= bn
                    If Len(Trim(body(jn))) > 0 Then Exit Do
                    jn = jn + 1
                Loop
                nx = ""
                If jn <= bn Then nx = LCase(Left(Trim(body(jn)), 4))
                badn = (nx = "else") Or (Left(nx, 3) = "end") Or (Left(nx, 4) = "unti") _
                    Or (Left(nx, 1) = "+") Or (Left(nx, 1) = "-") Or (Left(nx, 3) = "and") _
                    Or (Left(nx, 2) = "or") Or (Left(nx, 1) = ".")
                If Not badn Then
                    ncand = ncand + 1
                    If ncand > UBound(cand) Then ReDim Preserve cand(1 To ncand * 2)
                    cand(ncand) = k
                End If
            End If
        End If
    Next k
End Sub

Private Function L9_NameTaken(ByRef L() As String, ByVal n As Long, ByVal nm As String) As Boolean
    Dim k As Long
    L9_NameTaken = False
    For k = 1 To n
        If InStr(L(k), "procedure " & nm & ";") > 0 Then L9_NameTaken = True: Exit Function
    Next k
End Function

Private Function L9_RenameWord(ByVal s As String, ByVal oldw As String, ByVal neww As String) As String
    Dim out As String, q As Boolean, i As Long, c As String, j As Long, w As String, cc As String
    out = "": q = False: i = 1
    Do While i <= Len(s)
        c = Mid(s, i, 1)
        If c = "'" Then
            q = Not q: out = out & c: i = i + 1
        ElseIf (Not q) And ((c >= "a" And c <= "z") Or (c >= "A" And c <= "Z") Or c = "_") Then
            j = i
            Do While j <= Len(s)
                cc = Mid(s, j, 1)
                If (cc >= "a" And cc <= "z") Or (cc >= "A" And cc <= "Z") Or (cc >= "0" And cc <= "9") Or cc = "_" Then
                    j = j + 1
                Else
                    Exit Do
                End If
            Loop
            w = Mid(s, i, j - i)
            If StrComp(w, oldw, vbTextCompare) = 0 And (j > Len(s) Or Not ((Mid(s, j, 1) >= "a" And Mid(s, j, 1) <= "z") Or (Mid(s, j, 1) >= "A" And Mid(s, j, 1) <= "Z") Or (Mid(s, j, 1) >= "0" And Mid(s, j, 1) <= "9") Or Mid(s, j, 1) = "_")) Then
                out = out & neww
            Else
                out = out & w
            End If
            i = j
        ElseIf c = "/" And (Not q) And i < Len(s) And Mid(s, i + 1, 1) = "/" Then
            out = out & Mid(s, i)
            i = Len(s) + 1
        Else
            out = out & c: i = i + 1
        End If
    Loop
    L9_RenameWord = out
End Function

Private Function L9_FindDecl(ByRef L() As String, ByVal n As Long, ByVal nm As String) As Long
    Dim k As Long, t As String
    L9_FindDecl = 0
    For k = 1 To n
        t = Trim(L(k))
        If LCase(t) = "implementation" Then Exit For
        If LCase(Left(t, 10)) = "procedure " Then
            If Left(Mid(t, 11), Len(nm) + 1) = nm & ";" Then L9_FindDecl = k: Exit Function
        End If
    Next k
End Function

Private Sub L9_FindLoop(ByRef base1() As String, ByVal bn1 As Long, ByVal thrMono As Long, ByRef lbA As Long, ByRef lbE As Long)
    Dim i As Long, dd As Long, pv As Long, ltt As String
    Dim lws() As String, k As Long, lw As String, lhasB As Boolean, lhasE As Boolean
    Dim d2 As Long, j As Long
    lbA = 0: lbE = 0: dd = 0
    For i = 1 To bn1
        pv = dd
        dd = dd + L9_Delta(base1(i))
        If lbA = 0 Then
            ltt = LCase(L9_MonoStrip(base1(i)))
            If InStr(1, ltt, "for ") > 0 Or InStr(1, ltt, "while ") > 0 Then
                lhasB = False: lhasE = False
                lws = L9_MonoWords(ltt)
                For k = LBound(lws) To UBound(lws)
                    lw = lws(k)
                    If lw = "begin" Or lw = "repeat" Then lhasB = True
                    If lw = "end" Or lw = "until" Then lhasE = True
                Next k
                If lhasB And Not lhasE Then
                    d2 = dd
                    For j = i + 1 To bn1
                        d2 = d2 + L9_Delta(base1(j))
                        If d2 = pv Then
                            If (j - i - 1) > thrMono Then
                                lbA = i: lbE = j
                            End If
                            Exit For
                        End If
                    Next j
                    If lbA > 0 Then Exit For
                End If
            End If
        End If
    Next i
End Sub

Private Function L9_MonoSplit(ByRef base1() As String, ByRef bn1 As Long, _
    ByVal thrMono As Long, ByRef nParts As Long, ByRef maxPart As Long, _
    ByRef msg As String, ByRef cycFlag As Long, ByRef joints() As Long, ByRef nj As Long) As Boolean
    Dim i As Long, j As Long, k As Long, dd As Long, pv As Long
    Dim bA As Long, be As Long, blen As Long
    Dim ifStart As Long, t As String, tt As String
    Dim cvars() As String, nc As Long, ws() As String, w As String
    Dim badCond As Boolean, oneSided As Long, hasB As Boolean, hasE As Boolean, hasC As Boolean
    Dim cand() As Long, ncand As Long, jn As Long, nx As String, badn As Boolean
    Dim np As Long, tg As Long, bS As Long, lastc As Long
    Dim cts() As Long
    Dim out2() As String, m2 As Long, cap2 As Long, pE2 As Long
    L9_MonoSplit = False: nParts = 0: maxPart = 0: msg = "": cycFlag = 0: nj = 0
    bA = 0: be = 0: dd = 0
    For i = 1 To bn1
        pv = dd
        dd = dd + L9_Delta(base1(i))
        If pv = 0 And dd > 0 Then bA = i
        If dd = 0 And pv > 0 And bA > 0 Then
            If (i - bA - 1) > thrMono Then
                be = i: Exit For
            Else
                bA = 0: be = 0
            End If
        End If
    Next i
    If bA = 0 Then Exit Function
    blen = be - bA - 1
    ifStart = bA
    For i = bA - 1 To 1 Step -1
        t = LCase(Trim(base1(i)))
        If Len(t) = 0 Then
            ifStart = i
        ElseIf Left(t, 3) = "if " Or Left(t, 3) = "if(" Or Left(t, 4) = "elseif" Then
            ifStart = i: Exit For
        Else
            ifStart = bA: Exit For
        End If
    Next i
    nc = 0: ReDim cvars(1 To 16)
    For i = ifStart To bA
        ws = L9_MonoWords(L9_MonoStrip(base1(i)))
        For j = LBound(ws) To UBound(ws)
            w = ws(j)
            If Len(w) > 0 And w <> "if" And w <> "then" _
               And w <> "begin" _
               And w <> "and" And w <> "or" And w <> "not" And w <> "elseif" _
               And w <> "end" And w <> "=" Then
                If Not ((Left(w, 1) >= "0") And (Left(w, 1) <= "9")) Then
                    nc = nc + 1
                    If nc > UBound(cvars) Then ReDim Preserve cvars(1 To nc * 2)
                    cvars(nc) = w
                End If
            End If
        Next j
    Next i
    oneSided = 0: badCond = False: cycFlag = 0
    For i = bA + 1 To be - 1
        tt = LCase(L9_MonoStrip(base1(i)))
        ws = L9_MonoWords(tt)
        hasB = False: hasE = False: hasC = False
        For j = LBound(ws) To UBound(ws)
            w = ws(j)
            If w = "begin" Or w = "repeat" Then hasB = True
            If w = "end" Or w = "until" Then hasE = True
            If w = "for" Or w = "while" Or w = "case" Then hasC = True
        Next j
        If hasC And Not hasB And Not hasE Then oneSided = oneSided + 1
        If hasC And hasB And Not hasE Then cycFlag = 1
        If (Not badCond) And nc > 0 Then
            For j = 1 To nc
                If InStr(1, tt, cvars(j) & ":=") > 0 Then badCond = True: Exit For
                If InStr(1, tt, cvars(j) & " :=") > 0 Then badCond = True: Exit For
            Next j
        End If
    Next i
    If oneSided > 0 Then
        msg = "Ф32а: блок " & CStr(blen) & " НЕ РЕЗАН: одностороние цикл-строки (" & CStr(oneSided) & ")"
        Exit Function
    End If
    If badCond Then
        msg = "Ф32а: блок " & CStr(blen) & " НЕ РЕЗАН: переменная условия присваивается внутри блока"
        Exit Function
    End If
    If cycFlag = 1 Then
        msg = "Ф32а: блок " & CStr(blen) & " содержит многострочный цикл - путь Ф32б"
        Exit Function
    End If
    ncand = 0: ReDim cand(1 To 64)
    dd = 0
    For i = bA + 1 To be - 1
        t = base1(i)
        pv = dd
        dd = dd + L9_Delta(t)
        If dd = 0 And pv = 0 Then
            tt = Trim(t)
            If Left(tt, 2) = "//" Then
                ncand = ncand + 1
                If ncand > UBound(cand) Then ReDim Preserve cand(1 To ncand * 2)
                cand(ncand) = i - bA
            ElseIf Right(tt, 1) = ";" And LCase(Left(tt, 3)) <> "end" And LCase(Left(tt, 4)) <> "else" Then
                jn = i + 1
                Do While jn <= be - 1
                    If Len(Trim(base1(jn))) > 0 Then Exit Do
                    jn = jn + 1
                Loop
                nx = ""
                If jn <= be - 1 Then nx = LCase(Left(Trim(base1(jn)), 4))
                badn = (nx = "else") Or (Left(nx, 3) = "end") Or (Left(nx, 4) = "unti") _
                    Or (Left(nx, 1) = "+") Or (Left(nx, 1) = "-") Or (Left(nx, 3) = "and") _
                    Or (Left(nx, 2) = "or") Or (Left(nx, 1) = ".")
                If Not badn Then
                    ncand = ncand + 1
                    If ncand > UBound(cand) Then ReDim Preserve cand(1 To ncand * 2)
                    cand(ncand) = i - bA
                End If
            End If
        End If
    Next i
    If ncand = 0 Then
        msg = "Ф32а: блок " & CStr(blen) & " НЕ РЕЗАН: точек реза внутри блока нет"
        Exit Function
    End If
    np = (blen + thrMono - 1) \ thrMono
    If np < 2 Then np = 2
    ReDim cts(0 To np)
    lastc = 0
    For k = 1 To np - 1
        tg = k * blen \ np
        bS = 0
        For j = 1 To ncand
            If cand(j) > lastc Then
                If bS = 0 Or Abs(cand(j) - tg) < Abs(bS - tg) Then bS = cand(j)
            End If
        Next j
        If bS = 0 Then bS = blen - 1
        If bS >= blen Then bS = blen - 1
        If bS <= lastc Then bS = lastc + 1
        cts(k) = bS: lastc = bS
    Next k
    cts(np) = blen
    cap2 = bn1 + np * 4 + 16
    ReDim out2(1 To cap2): m2 = 0
    ReDim joints(1 To np): nj = 0
    For i = 1 To bA
        m2 = m2 + 1: out2(m2) = base1(i)
    Next i
    For k = 1 To np
        For i = bA + cts(k - 1) + 1 To bA + cts(k)
            m2 = m2 + 1: out2(m2) = base1(i)
        Next i
        If k < np Then
            pE2 = InStr(base1(be), ";")
            If pE2 > 0 Then
                m2 = m2 + 1: out2(m2) = Left(base1(be), pE2)
            Else
                m2 = m2 + 1: out2(m2) = base1(be)
            End If
            nj = nj + 1: joints(nj) = m2
            For i = ifStart To bA
                m2 = m2 + 1: out2(m2) = base1(i)
            Next i
        End If
    Next k
    For i = be To bn1
        m2 = m2 + 1: out2(m2) = base1(i)
    Next i
    ReDim Preserve base1(1 To m2)
    For i = 1 To m2
        base1(i) = out2(i)
    Next i
    bn1 = m2
    nParts = np
    maxPart = 0
    For k = 1 To np
        If (cts(k) - cts(k - 1)) > maxPart Then maxPart = cts(k) - cts(k - 1)
    Next k
    msg = "Ф32а: блок " & CStr(blen) & " -> " & CStr(np) & " кусков, макс. кусок " & CStr(maxPart) & " -> режет N-резку по стыкам"
    L9_MonoSplit = True
End Function

Private Sub L9_ApplyCut(ByRef L() As String, ByRef n As Long, ByVal cls As String, ByVal nm As String, _
    ByVal marker As String, ByRef names() As String, ByVal nP As Long, _
    ByRef bodyTxt() As String, ByRef inner2() As String, ByVal inn2 As Long, ByRef exc() As String, ByVal td1 As Long, _
    ByRef fldNm() As String, ByRef fldTp() As String, ByVal nfl As Long, ByRef outRes() As String, ByRef mRes As Long)
    Dim out() As String, m As Long, cap As Long
    cap = n + nP * 4 + 512
    ReDim out(1 To cap): m = 0
    Dim i As Long, j As Long, k As Long
    Dim hdr As Long, bi As Long
    hdr = 0: bi = 0
    For i = 1 To n
        If (Len(cls) > 0 And Trim(L(i)) = "procedure " & cls & "." & nm & ";") Or (Len(cls) = 0 And Trim(L(i)) = "procedure " & nm & ";") Then
            hdr = i: Exit For
        End If
    Next i
    If hdr = 0 Then Exit Sub
    For j = hdr + 1 To n
        If Trim(L(j)) = "begin" Then bi = j: Exit For
    Next j
    If bi = 0 Then Exit Sub
    Dim dd As Long, ei As Long
    dd = 1: ei = 0
    For j = bi + 1 To n
        dd = dd + L9_Delta(L(j))
        If dd <= 0 Then ei = j: Exit For
    Next j
    For i = 1 To bi
        m = m + 1: out(m) = L(i)
    Next i
    If td1 = 1 Then
        m = m + 1: out(m) = "  try"
    End If
    If inn2 > 0 Then
        For j = 1 To inn2
            m = m + 1: out(m) = inner2(j)
        Next j
    Else
        For k = 1 To nP
            m = m + 1: out(m) = "    " & names(k) & ";  // " & marker & ": часть 1 из " & CStr(nP) & " процедуры " & nm
        Next k
    End If
    For k = 1 To UBound(exc)
        m = m + 1: out(m) = exc(k)
    Next k
    m = m + 1: out(m) = "end;"
    For i = ei + 1 To n
        m = m + 1: out(m) = L(i)
    Next i
    Dim diNew As Long
    diNew = 0
    For i = 1 To m
        If LCase(Left(Trim(out(i)), 10)) = "procedure " Then
            If Left(Mid(Trim(out(i)), 11), Len(nm) + 1) = nm & ";" Then diNew = i: Exit For
        End If
    Next i
    If diNew > 0 Then
        Dim nl() As String, out2() As String, m2 As Long
        ReDim nl(1 To nP)
        For k = 1 To nP
            nl(k) = "    procedure " & names(k) & ";      // " & marker & ": часть " & CStr(k) & " из " & CStr(nP) & " процедуры " & nm
        Next k
        m2 = 0: cap = m + nP + 8
        ReDim out2(1 To cap)
        For i = 1 To diNew: m2 = m2 + 1: out2(m2) = out(i): Next i
        For k = 1 To nP: m2 = m2 + 1: out2(m2) = nl(k): Next k
        For i = diNew + 1 To m: m2 = m2 + 1: out2(m2) = out(i): Next i
        m = m2: out = out2
    End If
    If nfl > 0 Then
        Dim clsLn As Long
        clsLn = 0
        For i = 1 To m
            If InStr(LCase(out(i)), "=class(") > 0 Then clsLn = i: Exit For
        Next i
        If clsLn > 0 Then
            m2 = 0: cap = m + nfl + 8
            ReDim out2(1 To cap)
            For i = 1 To clsLn: m2 = m2 + 1: out2(m2) = out(i): Next i
            For k = 1 To nfl
                m2 = m2 + 1: out2(m2) = "    " & fldNm(k) & ": " & fldTp(k) & ";  // " & marker & ": локал процедуры " & nm
            Next k
            For i = clsLn + 1 To m: m2 = m2 + 1: out2(m2) = out(i): Next i
            m = m2: out = out2
        End If
    End If
    Dim hdr2 As Long, beg2 As Long, ei2 As Long, dd3 As Long
    hdr2 = 0
    For i = 1 To m
        If (Len(cls) > 0 And Trim(out(i)) = "procedure " & cls & "." & nm & ";") Or (Len(cls) = 0 And Trim(out(i)) = "procedure " & nm & ";") Then
            hdr2 = i: Exit For
        End If
    Next i
    If hdr2 > 0 Then
        For j = hdr2 + 1 To m
            If Trim(out(j)) = "begin" Then beg2 = j: Exit For
        Next j
        dd3 = 1: ei2 = 0
        For j = beg2 + 1 To m
            dd3 = dd3 + L9_Delta(out(j))
            If dd3 <= 0 Then ei2 = j: Exit For
        Next j
        If ei2 > 0 Then
            m2 = 0: cap = m + nP * 8 + 64
            ReDim out2(1 To cap)
            For i = 1 To ei2: m2 = m2 + 1: out2(m2) = out(i): Next i
            Dim lines2() As String
            For k = 1 To nP
                m2 = m2 + 1: out2(m2) = ""
                m2 = m2 + 1: out2(m2) = "procedure " & cls & "." & names(k) & ";"
                m2 = m2 + 1: out2(m2) = "begin"
                lines2 = Split(bodyTxt(k), vbCrLf)
                For j = LBound(lines2) To UBound(lines2)
                    If Len(lines2(j)) > 0 Then
