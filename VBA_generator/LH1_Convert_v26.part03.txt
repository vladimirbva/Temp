Private Sub L9_CutOne(ByRef L() As String, ByRef n As Long, ByVal cls As String, ByVal nm As String, _
    ByVal marker As String, ByRef nCut As Long, ByRef nPartsAll As Long, ByRef nMaxAll As Long, ByRef msgLog As String, ByVal fp As String)
    Const L9_THR_LINE As Long = 5000: Const L9_THR_INNER As Long = 4000   ' LH1 v26 Ф4: локальные
    Dim hdr As Long, bi As Long, ei As Long, declI As Long
    Dim i As Long, j As Long, k As Long, dd As Long, tt As String, pp As Long
    Dim out() As String, m As Long
    msgLog = ""
    declI = L9_FindDecl(L, n, nm)
    If declI = 0 Then msgLog = "skip: нет декларации": Exit Sub
    hdr = 0
    For i = 1 To n
        If (Len(cls) > 0 And Trim(L(i)) = "procedure " & cls & "." & nm & ";") Or (Len(cls) = 0 And Trim(L(i)) = "procedure " & nm & ";") Then
            hdr = i: Exit For
        End If
    Next i
    If hdr = 0 Then msgLog = "skip: нет реализации": Exit Sub
    bi = 0
    For j = hdr + 1 To n
        If Trim(L(j)) = "begin" Then bi = j: Exit For
    Next j
    If bi = 0 Then msgLog = "skip: нет begin": Exit Sub
    dd = 1: ei = 0
    For j = bi + 1 To n
        dd = dd + L9_Delta(L(j))
        If dd <= 0 Then ei = j: Exit For
    Next j
    If ei = 0 Then msgLog = "skip: нет конца": Exit Sub
    If (ei - hdr) <= L9_THR_LINE Then msgLog = "skip: в пороге": Exit Sub
    Dim bL() As String, bln As Long
    bln = ei - bi - 1
    ReDim bL(1 To bln)
    For j = 1 To bln: bL(j) = L(bi + j): Next j
    Dim idxExc As Long, dd2 As Long
    dd2 = 0: idxExc = 0
    For j = 1 To bln
        tt = Trim(bL(j))
        If dd2 = 1 And (tt = "except" Or Left(tt, 7) = "except ") Then idxExc = j: Exit For
        dd2 = dd2 + L9_Delta(bL(j))
    Next j
    Dim inner() As String, inn As Long, exc() As String, exn As Long
    inn = 0: exn = 0
    If idxExc > 0 Then
        ReDim inner(1 To idxExc - 1)
        For j = 1 To idxExc - 1: inn = inn + 1: inner(inn) = bL(j): Next j
        ReDim exc(1 To bln - idxExc + 1)
        For j = idxExc To bln: exn = exn + 1: exc(exn) = bL(j): Next j
    Else
        ReDim inner(1 To bln)
        For j = 1 To bln: inn = inn + 1: inner(inn) = bL(j): Next j
        ReDim exc(1 To 1): exc(1) = ""
    End If
    Dim td1 As Long
    td1 = 0
    If inn > 0 And (Trim(inner(1)) = "try" Or Left(Trim(inner(1)), 4) = "try ") Then td1 = 1
    Dim base1() As String, bn1 As Long
    If td1 = 1 Then
        bn1 = inn - 1
        ReDim base1(1 To bn1)
        For j = 2 To inn: base1(j - 1) = inner(j): Next j
    Else
        bn1 = inn
        ReDim base1(1 To bn1)
        For j = 1 To inn: base1(j) = inner(j): Next j
    End If
    If bn1 <= L9_THR_LINE Then msgLog = "skip: в пороге": Exit Sub
    Dim fldNm() As String, fldTp() As String, nfl As Long
    ReDim fldNm(1 To 16): ReDim fldTp(1 To 16): nfl = 0
    For j = hdr + 1 To bi - 1
        tt = Trim(L(j))
        If Len(tt) > 0 And LCase(tt) <> "var" And Left(tt, 2) <> "//" And Left(tt, 1) <> "{" Then
            Dim lhs As String, tp As String, vv() As String, vi As Long, vn As String
            pp = InStr(tt, ":")
            If pp > 1 Then
                lhs = Trim(Left(tt, pp - 1))
                tp = Trim(Mid(tt, pp + 1))
                If Right(tp, 1) = ";" Then tp = Trim(Left(tp, Len(tp) - 1))
                vv = Split(lhs, ",")
                For vi = LBound(vv) To UBound(vv)
                    vn = Trim(vv(vi))
                    If Len(vn) > 0 And L9_Simple(vn) Then
                        nfl = nfl + 1
                        If nfl > UBound(fldNm) Then
                            ReDim Preserve fldNm(1 To nfl * 2): ReDim Preserve fldTp(1 To nfl * 2)
                        End If
                        fldNm(nfl) = nm & "_" & vn
                        fldTp(nfl) = tp
                        For k = 1 To bn1: base1(k) = L9_RenameWord(base1(k), vn, fldNm(nfl)): Next k
                        For k = 1 To exn: exc(k) = L9_RenameWord(exc(k), vn, fldNm(nfl)): Next k
                    End If
                Next vi
            End If
        End If
    Next j
    Dim lbA As Long, lbE As Long
    L9_FindLoop base1, bn1, L9_THR_INNER, lbA, lbE
    Dim names() As String, nP As Long
    Dim bodyTxt() As String
    Dim inner2() As String, inn2 As Long
    Dim note As String, maxp As Long
    Dim qn As String, seg As String, seg2 As String
    If lbA > 0 Then
        Dim llen As Long
        llen = lbE - lbA - 1
        Dim tbody() As String
        ReDim tbody(1 To llen)
        For j = 1 To llen: tbody(j) = base1(lbA + j): Next j
        Dim tcand() As Long, tncand As Long
        ReDim tcand(1 To 64)
        L9_CutCand tbody, llen, tcand, tncand, 0
        nP = (llen + L9_THR_INNER - 1) \ L9_THR_INNER
        If nP < 2 Then nP = 2
        Dim lcts() As Long, llast As Long, ltg As Long, lbs As Long
        ReDim lcts(0 To nP): llast = 0
        For k = 1 To nP - 1
            ltg = k * llen \ nP
            lbs = 0
            For j = 1 To tncand
                If tcand(j) > llast Then
                    If lbs = 0 Or Abs(tcand(j) - ltg) < Abs(lbs - ltg) Then lbs = tcand(j)
                End If
            Next j
            If lbs = 0 Then lbs = llen - 1
            If lbs >= llen Then lbs = llen - 1
            If lbs <= llast Then lbs = llast + 1
            lcts(k) = lbs: llast = lbs
        Next k
        lcts(nP) = llen
        ReDim names(1 To nP)
        For k = 1 To nP
            qn = nm & L9_Suf(k)
            Do While L9_NameTaken(L, n, qn)
                qn = qn & "x"
            Loop
            names(k) = qn
        Next k
        ReDim inner2(1 To bn1 + nP + 8): inn2 = 0
        For j = 1 To lbA + 1
            inn2 = inn2 + 1: inner2(inn2) = base1(j)
        Next j
        For k = 1 To nP
            inn2 = inn2 + 1: inner2(inn2) = "    " & names(k) & ";  // " & marker & ": часть " & CStr(k) & " из " & CStr(nP) & " цикла " & nm
        Next k
        For j = lbE To bn1
            inn2 = inn2 + 1: inner2(inn2) = base1(j)
        Next j
        ReDim Preserve inner2(1 To inn2)
        maxp = 0
        ReDim bodyTxt(1 To nP)
        For k = 1 To nP
            seg = ""
            For j = lcts(k - 1) + 1 To lcts(k)
                seg = seg & tbody(j) & vbCrLf
            Next j
            bodyTxt(k) = seg
            If (lcts(k) - lcts(k - 1)) > maxp Then maxp = lcts(k) - lcts(k - 1)
        Next k
        note = "Ф32б: цикл " & CStr(llen) & " -> " & CStr(nP) & " частей, макс. часть " & CStr(maxp)
    Else
        Dim joints() As Long, nj As Long
        Dim monoApplied As Boolean, monoParts As Long, monoMax As Long, monoMsg As String, monoCyc As Long
        monoApplied = L9_MonoSplit(base1, bn1, L9_THR_INNER, monoParts, monoMax, monoMsg, monoCyc, joints, nj)
        If Not monoApplied Then
            msgLog = "skip: " & monoMsg
            Exit Sub
        End If
        Dim cand() As Long, ncand As Long
        ReDim cand(1 To 64)
        L9_CutCand base1, bn1, cand, ncand, 0
        For k = 1 To nj
            ncand = ncand + 1
            If ncand > UBound(cand) Then ReDim Preserve cand(1 To ncand * 2)
            cand(ncand) = joints(k)
        Next k
        Dim n As Long
        n = bn1
        If n <= L9_THR_LINE Then
            msgLog = "skip: в пороге после Ф32а"
            Exit Sub
        End If
        nP = (n + L9_THR_LINE - 1) \ L9_THR_LINE
        If nP < 2 Then nP = 2
        Dim cts() As Long, last As Long, tg As Long, bs As Long
        ReDim cts(0 To nP): last = 0
        For k = 1 To nP - 1
            tg = k * n \ nP
            bs = 0
            For j = 1 To ncand
                If cand(j) > last Then
                    If bs = 0 Or Abs(cand(j) - tg) < Abs(bs - tg) Then bs = cand(j)
                End If
            Next j
            If bs = 0 Then bs = n - 1
            If bs >= n Then bs = n - 1
            If bs <= last Then bs = last + 1
            cts(k) = bs: last = bs
        Next k
        cts(nP) = n
        ReDim names(1 To nP)
        For k = 1 To nP
            qn = nm & L9_Suf(k)
            Do While L9_NameTaken(L, n, qn)
                qn = qn & "x"
            Loop
            names(k) = qn
        Next k
        maxp = 0
        ReDim bodyTxt(1 To nP)
        For k = 1 To nP
            seg2 = ""
            For j = cts(k - 1) + 1 To cts(k)
                seg2 = seg2 & base1(j) & vbCrLf
            Next j
            bodyTxt(k) = seg2
            If (cts(k) - cts(k - 1)) > maxp Then maxp = cts(k) - cts(k - 1)
        Next k
        ReDim inner2(1 To 1): inn2 = 0
        note = "Ф32а+N-резка: " & CStr(nP) & " частей, макс. часть " & CStr(maxp)
    End If
    L9_ApplyCut L, n, cls, nm, marker, names, nP, bodyTxt, inner2, inn2, exc, td1, fldNm, fldTp, nfl, out, m
    n = m
    Dim f2 As Integer
    f2 = FreeFile
    Open fp For Output As #f2
    For i = 1 To n
        Print #f2, out(i)
    Next i
    Close #f2
    nCut = nCut + 1
    nPartsAll = nPartsAll + nP
    If maxp > nMaxAll Then nMaxAll = maxp
    msgLog = note
End Sub

Private Sub L9_CutFile(ByVal pth As String, ByVal fn As String, ByRef nCut As Long, ByRef nPartsAll As Long, ByRef nMaxAll As Long, ByVal marker As String)
    Const L9_THR_LINE As Long = 5000   ' LH1 v26 Ф4: локальные - вставка самодостаточна
    Dim fpath As String, f1 As Integer
    Dim L() As String, n As Long, ln As String
    Dim i As Long, k As Long, e As Long
    Dim prs() As Long, npr As Long
    Dim skipFrom As Long, it As Long
    Dim cls As String, nm As String, msgLog As String
    Dim tgt As Long
    fpath = pth & fn
    f1 = FreeFile
    ReDim L(1 To 4096): n = 0
    Open fpath For Input As #f1
    Do While Not EOF(f1)
        If n = UBound(L) Then ReDim Preserve L(1 To UBound(L) * 2)
        n = n + 1
        Line Input #f1, ln
        If InStr(ln, "Ф32-пост-резка") > 0 Then
            Close #f1
            Exit Sub
        End If
        L(n) = ln
    Loop
    Close #f1
    ReDim Preserve L(1 To n)
    skipFrom = 0
    For it = 1 To 200
        ReDim L(1 To 4096): n = 0
        f1 = FreeFile
        Open fpath For Input As #f1
        Do While Not EOF(f1)
            If n = UBound(L) Then ReDim Preserve L(1 To UBound(L) * 2)
            n = n + 1
            Line Input #f1, ln
            L(n) = ln
        Loop
        Close #f1
        ReDim Preserve L(1 To n)
        ReDim prs(1 To n \ 8 + 8): npr = 0
        For i = 1 To n
            If Left(Trim(L(i)), 10) = "procedure " Then
                If npr = UBound(prs) Then ReDim Preserve prs(1 To npr * 2)
                npr = npr + 1: prs(npr) = i
            End If
        Next i
        tgt = 0
        For k = 1 To npr
            e = n
            If k < npr Then e = prs(k + 1) - 1
            If e - prs(k) > L9_THR_LINE And prs(k) > skipFrom Then
                nm = Trim(L(prs(k)))
                nm = Left(nm, Len(nm) - 1)
                If InStr(nm, "(") > 0 Then nm = Trim(Left(nm, InStr(nm, "(") - 1))
                cls = ""
                If InStr(nm, ".") > 0 Then cls = Left(nm, InStr(nm, ".") - 1): nm = Mid(nm, InStr(nm, ".") + 1)
                tgt = prs(k)
                Exit For
            End If
        Next k
        If tgt = 0 Then Exit For
        L9_CutOne L, n, cls, nm, marker, nCut, nPartsAll, nMaxAll, msgLog, fpath
        If Left(msgLog, 4) = "skip" Then
            skipFrom = tgt
        Else
            skipFrom = 0
        End If
    Next it
    f1 = FreeFile
    Open fpath For Append As #f1
    Print #f1, "// Ф32-пост-резка " & marker & ": процедур разрезано " & CStr(nCut) & ", частей добавлено " & CStr(nPartsAll) & ", макс. часть " & CStr(nMaxAll)
    Close #f1
End Sub
