Option Explicit
Private csh1 As Boolean

' --- простой идентификатор (безопасно гасить присваиванием) ---
Private Function L1_Simple(ByVal s As String) As Boolean
    Dim i As Long, c As String
    L1_Simple = False
    If Len(s) = 0 Then Exit Function
    For i = 1 To Len(s)
        c = Mid(s, i, 1)
        If Not ((c >= "a" And c <= "z") Or (c >= "A" And c <= "Z") Or (c >= "0" And c <= "9") Or c = "_") Then Exit Function
    Next i
    L1_Simple = True
End Function
' --- дельта глубины begin/try/repeat/case - end/until (вне строк-литералов) ---
Private Function L1_Delta(ByVal s As String) As Long
    Dim q As Boolean, i As Long, c As String, buf As String, t As String
    Dim d As Long, j As Long, w As String
    q = False: buf = ""
    For i = 1 To Len(s)
        c = Mid(s, i, 1)
        If c = "'" Then
            q = Not q: buf = buf & c
        ElseIf c = "/" And (Not q) And i < Len(s) And Mid(s, i + 1, 1) = "/" Then
            Exit For
        Else
            buf = buf & c
        End If
    Next i
    t = LCase(buf)
    If InStr(t, "begin") = 0 And InStr(t, "end") = 0 And InStr(t, "try") = 0 _
       And InStr(t, "repeat") = 0 And InStr(t, "until") = 0 And InStr(t, "case") = 0 Then
        L1_Delta = 0: Exit Function
    End If
    For i = 1 To Len(t)
        c = Mid(t, i, 1)
        If Not ((c >= "a" And c <= "z") Or (c >= "0" And c <= "9") Or c = "_") Then Mid(t, i, 1) = " "
    Next i
    d = 0
    Dim parts() As String
    parts = Split(t, " ")
    For j = LBound(parts) To UBound(parts)
        w = parts(j)
        If w = "begin" Or w = "try" Or w = "repeat" Then d = d + 1
        If w = "end" Or w = "until" Then d = d - 1
        If w = "case" Then d = d + 1
    Next j
    L1_Delta = d
End Function
