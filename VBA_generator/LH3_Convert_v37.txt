' LH3 v26 (Ф36.2): канарейка при старте конвертации.
' PASS-условия: (1) тождество Power(2.7182,0.23)=Exp(0.23*Log(2.7182)) с точн. 1e-9 отн.
'   (эталон - Log в рантайме, Ф41 v28; замена корректна по определению LNG2E=ln(2.7182);
'   + ПРОВЕРКА 1б: |LNG2E - Log(2.7182)| <= 1e-8 абс. - ловит подмену константы);
'   (2) критерий основания: L1_IsEBase("2.7182")=True, L1_IsEBase("2.72")=False,
'   L1_IsEBase("((po_p+1)/(pi_p+1))")=False (критерий не расширен);
'   (3) контрпример: Power(2,0.23)=1.1726 НЕ равен Exp(0.23*LNG2E)=1.5838 (расхождение есть).
' FAIL -> MsgBox + Exit Sub (конвертация не выпускается) - как Ф37.4 в CG4 v12.
Private Function L1_CanaryPowers() As Boolean
    Dim lhs As Double, rhs As Double
    L1_CanaryPowers = False
    ' ПРОВЕРКА 1 (Ф41 v28): эталон - Log(2.7182) в рантайме, а не округлённая константа
    lhs = (2.7182 ^ 0.23): rhs = Exp(0.23 * Log(2.7182))
    If Abs(lhs - rhs) > 0.000000001 * lhs Then Exit Function
    ' ПРОВЕРКА 1б (Ф41 v28): эмитируемая LNG2E точна до 1e-8 абс. (ловит подмену константы,
    ' не ловя её собственную погрешность 5.04e-9)
    If Abs(0.9999698915 - Log(2.7182)) > 0.00000001 Then Exit Function
    If Not L1_IsEBase("2.7182") Then Exit Function
    If L1_IsEBase("2.72") Then Exit Function
    If L1_IsEBase("((po_p+1)/(pi_p+1))") Then Exit Function
    lhs = (2# ^ 0.23): rhs = Exp(0.23 * 0.9999698915)
    If Abs(lhs - rhs) < 0.000001 * lhs Then Exit Function
    L1_CanaryPowers = True
End Function

' --- есть ли кириллица ВНЕ строк-литералов и вне "//" (Ф4 v2, перенос из LH1/LH2) ---
Private Function L1_CyrOut(ByVal s As String) As Boolean
    Dim q As String, i As Long, c As String, a As Long
    q = ""
    For i = 1 To Len(s)
        c = Mid(s, i, 1)
        If q <> "" Then
            If c = q Then q = ""
        ElseIf c = "'" Or c = Chr(34) Then
            q = c
        ElseIf c = "/" And i < Len(s) And Mid(s, i + 1, 1) = "/" Then
            Exit For
        Else
            a = Asc(c)
            If (a >= 192 And a <= 255) Or a = 168 Or a = 184 Then
                L1_CyrOut = True: Exit Function
            End If
        End If
    Next i
    L1_CyrOut = False
End Function

    Dim d As Long, j As Long, w As String
    q = False: buf = ""

    For i = 1 To Len(t)

' --- Power-фиксы v4 (С0-2, С1а-1, С1а-2): балансный разбор, скобки сохраняются ---
Private Function L1_FixPowers(ByVal s As String) As String
    Dim out As String, i As Long, p As Long, depth As Long, c As String
    Dim rest As String, args As String, j As Long, a1 As String, a2 As String
    out = "": i = 1
    Do While i <= Len(s)
        p = InStr(i, s, "Power(", vbTextCompare)
        If p = 0 Then out = out & Mid(s, i): Exit Do
        If p > 1 Then out = out & Mid(s, i, p - i)
        ' разбор скобок от p+5
        depth = 1: j = p + 6: args = ""
        Do While j <= Len(s) And depth > 0
            c = Mid(s, j, 1)
            If c = "(" Then depth = depth + 1
            If c = ")" Then
                depth = depth - 1
                If depth = 0 Then Exit Do
            End If
            args = args & c: j = j + 1
        Loop
        If depth = 0 Then
            ' args = A,B (A может быть выражением со скобками)
            depth = 0: a1 = "": a2 = ""
            For i = 1 To Len(args)
                c = Mid(args, i, 1)
                If c = "(" Then depth = depth + 1
                If c = ")" Then depth = depth - 1
                If c = "," And depth = 0 Then
                    a1 = Left(args, i - 1): a2 = Mid(args, i + 1): Exit For
                End If
            Next i
            a1 = Trim(a1): a2 = Trim(a2)
            If a1 <> "" And a2 <> "" Then
                If a2 = "2" Then
                    out = out & "(" & a1 & ")*(" & a1 & ")"          ' С0-2
                ElseIf a2 = "0.5" Then
                    out = out & "Sqrt(" & a1 & ")"                    ' С1а-2
                ElseIf LCase(Left(a2, 1)) >= "a" Then
                    out = out & "Power(" & args & ")"                     ' С1а-1 v26 (Ф36.1): буквенный показатель - НЕ заменять (v4-v25 теряли a2)
                    L1_cnt_POWFIX = L1_cnt_POWFIX + 1
                Else
                    If L1_IsEBase(a1) Then
                        out = out & "Exp((" & a1 & ")*" & a2 & "*LH3_LNG2E)"   ' С1а-1 v26 (Ф36.1): основание 2.7182 - корректная замена (Ln(2.7182)=LNG2E)
                        L1_cnt_EXP2 = L1_cnt_EXP2 + 1
                    Else
                        out = out & "Power(" & args & ")"                      ' С1а-1 v26 (Ф36.1): прочие основания - Power как есть (Ln(a) не подменяется)
                        L1_cnt_POWFIX = L1_cnt_POWFIX + 1
                    End If
                End If
            Else
                out = out & "Power(" & args & ")"
            End If
            i = j + 1
        Else
            out = out & "Power(": i = p + 6
        End If
    Loop
    L1_FixPowers = out
End Function

    If LCase(Left(s, 10)) = "procedure " Then
        s = Mid(s, 11)
    ElseIf LCase(Left(s, 9)) = "function " Then

Private Function L2_ClassName(ByRef lines() As String, ByVal impl As Long) As String
    Dim i As Long, t As String, p As Long
    L2_ClassName = "T_Hmodel_UKR"
    For i = 1 To impl - 1
        t = Trim(lines(i))
        p = InStr(LCase(t), "=class(")
        If p > 1 Then
            Dim w As String: w = Trim(Left(t, p - 1))
            If L2_Simple(w) Then L2_ClassName = w: Exit Function
        End If
    Next i
End Function

' --- есть ли кириллица ВНЕ строк-литералов и вне "//" (Ф5, перенос Ф4 из LH1) ---
Private Sub L2_CollectForVars(ByVal s As String, ByRef forSet As Object)
    Dim p As Long, e As Long, q2 As Long, nm2 As String
    p = 1
    Do While p > 0 And p <= Len(s)
        p = InStr(p, LCase(s), "for ")
        If p = 0 Then Exit Do
        q2 = p + 4
        Do While q2 <= Len(s) And Mid(s, q2, 1) = " "
            q2 = q2 + 1
        Loop
        e = q2
        Do While e <= Len(s)
            If InStr("abcdefghijklmnopqrstuvwxyz0123456789_", LCase(Mid(s, e, 1))) = 0 Then Exit Do
            e = e + 1
        Loop
        nm2 = LCase(Mid(s, q2, e - q2))
        If Len(nm2) > 0 Then
            If Not forSet.Exists(nm2) Then forSet.Add nm2, 1
        End If
        p = e
    Loop
End Sub

Private Sub L2_PushL(ByRef out() As String, ByRef m As Long, ByRef cap As Long, ByVal s As String)
    m = m + 1
    If m > cap Then cap = cap * 2: ReDim Preserve out(1 To cap)
    out(m) = s
End Sub

Private Sub L2_PushR(ByRef out() As String, ByRef m As Long, ByRef cap As Long, ByRef arr() As String, ByVal a As Long, ByVal z As Long)
    Dim i As Long
    For i = a To z
        L2_PushL out, m, cap, arr(i)
    Next i
End Sub

' --- замена номера слота в гейте "if (Miss_ind[ N]<4) then" ---
Private Function L2_SlotSub(ByVal s As String, ByVal slot As Long) As String
    Dim p1 As Long, p2 As Long, num As String
    p1 = InStr(s, "Miss_ind[")
    If p1 = 0 Then L2_SlotSub = s: Exit Function
    p2 = InStr(p1, s, "]")
    If p2 <= p1 Then L2_SlotSub = s: Exit Function
    num = Trim(Mid(s, p1 + 9, p2 - p1 - 9))
    L2_SlotSub = Left(s, p1 + 8) & CStr(slot) & Mid(s, p2)
End Function

' --- замена номера слота во ВСЕХ рег-строках блока (Miss_proc/Miss_ind/Miss_kerr/Miss_proc_2) ---
Private Function L2_SlotAll(ByVal s As String, ByVal newS As Long) As String
    Dim r As String, p As Long, q As Long, keys As Variant, i As Long
    r = s
    keys = Array("Miss_proc[", "Miss_ind[", "Miss_kerr[", "Miss_proc_2[")
    For i = LBound(keys) To UBound(keys)
        p = InStr(r, keys(i))
        Do While p > 0
            q = InStr(p, r, "]")
            If q > p Then r = Left(r, p + Len(keys(i)) - 1) & CStr(newS) & Mid(r, q)
            p = InStr(p + 1, r, keys(i))
        Loop
    Next i
    L2_SlotAll = r
End Function

' --- максимальное тело процедуры в файле (исходник) ---
Private Function L2_MaxSourceBody(ByVal fPath As String) As Long
    Dim fn As Integer, n As Long, cap As Long, ln As String
    Dim lines() As String, i As Long, k As Long, t As String
    Dim np As Long, mx As Long, jb As Long, ke As Long, dd As Long, okk As Boolean
    L2_MaxSourceBody = 0
    If Len(Dir(fPath)) = 0 Then Exit Function
    fn = FreeFile
    Open fPath For Input As #fn
    n = 0: cap = 1000: ReDim lines(1 To cap)
    Do While Not EOF(fn)
        Line Input #fn, ln
        n = n + 1
        If n > cap Then cap = cap * 2: ReDim Preserve lines(1 To cap)
        lines(n) = ln
    Loop
    Close #fn
    ReDim Preserve lines(1 To n)
    Dim impl As Long: impl = 0
    For i = 1 To n
        If LCase(Trim(lines(i))) = "implementation" Then impl = i: Exit For
    Next i
    If impl = 0 Then Exit Function
    np = 0: mx = 0
    i = impl
    Do While i <= n
        t = Trim(lines(i))
        If (Left(LCase(t), 10) = "procedure " Or Left(LCase(t), 9) = "function ") And Right(t, 1) = ";" Then
            Dim nm As String: nm = L2_ProcName(t)
            jb = 0
            If Len(nm) > 0 Then
                Dim j As Long: j = i + 1
                Do While j <= n
                    If Trim(lines(j)) = "begin" Then jb = j: Exit Do
                    Dim t2 As String: t2 = Trim(lines(j))
                    If Left(LCase(t2), 10) = "procedure " Or Left(LCase(t2), 9) = "function " Then Exit Do
                    j = j + 1
                Loop
                If jb > i Then
                    ke = 0: okk = False: dd = 1
                    For k = jb + 1 To n
                        dd = dd + L2_Delta(lines(k))
                        If dd <= 0 Then ke = k: okk = True: Exit For
                    Next k
                    If okk Then
                        np = np + 1
                        If ke - jb > mx Then mx = ke - jb
                        i = ke + 1
                    Else
                        i = i + 1
                    End If
                Else
                    i = i + 1
                End If
            Else
                i = i + 1
            End If
        Else
            i = i + 1
        End If
    Loop
    L2_MaxSourceBody = mx
End Function

' --- точка разреза: коммент-строка или завершённая ';'-строка на нулевой глубине,
'     за которой не начинается продолжение выражения ---
' --- точка разреза: коммент-строка или завершённая ';'-строка на глубине depth,
'     за которой не начинается продолжение выражения ---
Private Sub L2_CutCandidates(ByRef body() As String, ByVal bn As Long, ByRef cand() As Long, ByRef ncand As Long, ByVal depth As Long)
    Dim k As Long, dd As Long, t As String, jn As Long, nx As String, badn As Boolean
    ncand = 0: dd = 0
    For k = 1 To bn
        dd = dd + L2_Delta(body(k))
        If dd = depth Then
            t = Trim(body(k))
            If Left(t, 2) = "//" Then
                ncand = ncand + 1
                If ncand > UBound(cand) Then ReDim Preserve cand(1 To ncand * 2)
                cand(ncand) = k
            ElseIf Right(t, 1) = ";" And LCase(Left(t, 3)) <> "end" And LCase(Left(t, 4)) <> "else" Then
                jn = k + 1
                Do While jn <= bn
                    If Len(Trim(body(jn))) > 0 Then Exit Do
                    jn = jn + 1
                Loop
                nx = ""
                If jn <= bn Then nx = LCase(Left(Trim(body(jn)), 4))
                badn = (nx = "else") Or (Left(nx, 3) = "end") Or (Left(nx, 4) = "unti") _
                    Or (Left(nx, 1) = "+") Or (Left(nx, 1) = "-") Or (Left(nx, 3) = "and") _
                    Or (Left(nx, 2) = "or") Or (Left(nx, 1) = ".")
                If Not badn Then
                    ncand = ncand + 1
                    If ncand > UBound(cand) Then ReDim Preserve cand(1 To ncand * 2)
                    cand(ncand) = k
                End If
            End If
        End If
    Next k
End Sub

' --- суффикс части: a..z, затем za, zb, ... ---
Private Function L2_Suf(ByVal k As Long) As String
    If k <= 26 Then
        L2_Suf = Chr(96 + k)
    Else
        L2_Suf = "z" & CStr(k - 26)
    End If
End Function


' --- ближайшая к середине точка из списка кандидатов ---
Private Function L2_BestCut(ByRef cand() As Long, ByVal ncand As Long, ByVal half As Double) As Long
    Dim k As Long, bst As Long
    bst = 0
    For k = 1 To ncand
        If cand(k) > 0 And cand(k) < half * 2 Then
            If bst = 0 Or Abs(cand(k) - half) < Abs(bst - half) Then bst = cand(k)
        End If
    Next k
    L2_BestCut = bst
End Function

Private Function L2_MonoStrip(ByVal s As String) As String
    Dim i As Long, q2 As Boolean, r As String, c As String
    q2 = False: r = ""
    For i = 1 To Len(s)
        c = Mid(s, i, 1)
        If c = "'" Then
            q2 = Not q2
        ElseIf c = "/" And (Not q2) And i < Len(s) And Mid(s, i + 1, 1) = "/" Then
            Exit For
        ElseIf Not q2 Then
            r = r & c
        End If
    Next i
    L2_MonoStrip = r
End Function

Private Function L2_MonoWords(ByVal s As String) As String()
    Dim r() As String
    Dim i As Long, c As String, buf As String, n As Long
    For i = 1 To Len(s)
        c = Mid(s, i, 1)
        If (c >= "a" And c <= "z") Or (c >= "0" And c <= "9") Or c = "_" Then
            buf = buf & c
        Else
            If Len(buf) > 0 Then
                n = n + 1
                ReDim Preserve r(1 To n)
                r(n) = buf
                buf = ""
            End If
        End If
    Next i
    If Len(buf) > 0 Then
        n = n + 1
        ReDim Preserve r(1 To n)
        r(n) = buf
    End If
    If n = 0 Then
        ReDim r(1 To 1)
        r(1) = ""
    End If
    L2_MonoWords = r
End Function

'=====================================================================
' Ф32а v21: в-блочная резка монолитного блока с дублированием условия.
' Возвращает True, если резала. Если в блоке многострочный цикл - НЕ режет,
' ставит cycFlag=1 (для Ф32б). Правила и стражи - см. шапку v21.
'=====================================================================
Private Function L2_MonoSplit(ByRef base1() As String, ByRef bn1 As Long, _
    ByVal thrMono As Long, ByRef nParts As Long, ByRef maxPart As Long, _
    ByRef msg As String, ByRef cycFlag As Long) As Boolean
    Dim i As Long, j As Long, k As Long, dd As Long, pv As Long
    Dim bA As Long, be As Long, blen As Long
    Dim ifStart As Long, t As String, tt As String
    Dim condTxt() As String, nCond As Long
    Dim cvars() As String, nc As Long, ws() As String, w As String
    Dim badCond As Boolean, oneSided As Long, hasB As Boolean, hasE As Boolean, hasC As Boolean
    Dim cand() As Long, ncand As Long, jn As Long, nx As String, badn As Boolean
    Dim np As Long, tg As Long, bS As Long, lastc As Long
    Dim cts() As Long
    Dim out2() As String, m2 As Long, cap2 As Long, pLen As Long, pE2 As Long
    L2_MonoSplit = False: nParts = 0: maxPart = 0: msg = "": cycFlag = 0
    bA = 0: be = 0: dd = 0
    For i = 1 To bn1
        pv = dd
        dd = dd + L2_Delta(base1(i))
        If pv = 0 And dd > 0 Then bA = i
        If dd = 0 And pv > 0 And bA > 0 Then
            If (i - bA - 1) > thrMono Then
                be = i: Exit For
            Else
                bA = 0: be = 0
            End If
        End If
    Next i
    If bA = 0 Then Exit Function
    blen = be - bA - 1
    ifStart = bA
    For i = bA - 1 To 1 Step -1
        t = LCase(Trim(base1(i)))
        If Len(t) = 0 Then
            ifStart = i
        ElseIf Left(t, 3) = "if " Or Left(t, 3) = "if(" Or Left(t, 4) = "elseif" Then
            ifStart = i: Exit For
        Else
            ifStart = bA: Exit For
        End If
    Next i
    nCond = bA - ifStart + 1
    ReDim condTxt(1 To nCond)
    For i = ifStart To bA
        condTxt(i - ifStart + 1) = base1(i)
    Next i
    nc = 0: ReDim cvars(1 To 16)
    For i = ifStart To bA
        ws = L2_MonoWords(L2_MonoStrip(base1(i)))
        For j = LBound(ws) To UBound(ws)
            w = ws(j)
            If Len(w) > 0 And w <> "if" And w <> "then" And w <> "begin" _
               And w <> "and" And w <> "or" And w <> "not" And w <> "elseif" _
               And w <> "end" And w <> "=" Then
                If Not ((Left(w, 1) >= "0") And (Left(w, 1) <= "9")) Then
                    nc = nc + 1
                    If nc > UBound(cvars) Then ReDim Preserve cvars(1 To nc * 2)
                    cvars(nc) = w
                End If
            End If
        Next j
    Next i
    oneSided = 0: badCond = False
    For i = bA + 1 To be - 1
        tt = LCase(L2_MonoStrip(base1(i)))
        ws = L2_MonoWords(tt)
        hasB = False: hasE = False: hasC = False
        For j = LBound(ws) To UBound(ws)
            w = ws(j)
            If w = "begin" Or w = "repeat" Then hasB = True
            If w = "end" Or w = "until" Then hasE = True
            If w = "for" Or w = "while" Or w = "case" Then hasC = True
        Next j
        If hasC And Not hasB And Not hasE Then oneSided = oneSided + 1
        If hasC And hasB And Not hasE Then cycFlag = 1
        If (Not badCond) And nc > 0 Then
            For j = 1 To nc
                If InStr(1, tt, cvars(j) & ":=") > 0 Then
                    badCond = True
                    Exit For
                End If
                If InStr(1, tt, cvars(j) & " :=") > 0 Then
                    badCond = True
                    Exit For
                End If
            Next j
        End If
    Next i
    If oneSided > 0 Then
        msg = "Ф32 v21: блок длиной " & CStr(blen) & " НЕ РЕЗАН: одностороние цикл-строки (" & CStr(oneSided) & ") - резка могла бы разорвать цикл"
        Exit Function
    End If
    If badCond Then
        msg = "Ф32 v21: блок длиной " & CStr(blen) & " НЕ РЕЗАН: переменная условия присваивается внутри блока (дублирование условия небезопасно)"
        Exit Function
    End If
    If cycFlag = 1 Then
        msg = "Ф32 v21: блок длиной " & CStr(blen) & " содержит многострочный цикl - в-блочная резка с дублированием условия НЕЛЬЗЯ, путь Ф32б (вынос тела цикла)"
        Exit Function
    End If
    ncand = 0: ReDim cand(1 To 64)
    dd = 0
    For i = bA + 1 To be - 1
        t = base1(i)
        pv = dd
        dd = dd + L2_Delta(t)
        If dd = 0 And pv = 0 Then
            tt = Trim(t)
            If Left(tt, 2) = "//" Then
                ncand = ncand + 1
                If ncand > UBound(cand) Then ReDim Preserve cand(1 To ncand * 2)
                cand(ncand) = i - bA
            ElseIf Right(tt, 1) = ";" And LCase(Left(tt, 3)) <> "end" And LCase(Left(tt, 4)) <> "else" Then
                jn = i + 1
                Do While jn <= be - 1
                    If Len(Trim(base1(jn))) > 0 Then Exit Do
                    jn = jn + 1
                Loop
                nx = ""
                If jn <= be - 1 Then nx = LCase(Left(Trim(base1(jn)), 4))
                badn = (nx = "else") Or (Left(nx, 3) = "end") Or (Left(nx, 4) = "unti") _
                    Or (Left(nx, 1) = "+") Or (Left(nx, 1) = "-") Or (Left(nx, 3) = "and") _
                    Or (Left(nx, 2) = "or") Or (Left(nx, 1) = ".")
                If Not badn Then
                    ncand = ncand + 1
                    If ncand > UBound(cand) Then ReDim Preserve cand(1 To ncand * 2)
                    cand(ncand) = i - bA
                End If
            End If
        End If
    Next i
    If ncand = 0 Then
        msg = "Ф32 v21: блок длиной " & CStr(blen) & " НЕ РЕЗАН: точек реза внутри блока нет"
        Exit Function
    End If
    np = (blen + thrMono - 1) \ thrMono
    If np < 2 Then np = 2
    ReDim cts(0 To np)
    lastc = 0
    For k = 1 To np - 1
        tg = k * blen \ np
        bS = 0
        For j = 1 To ncand
            If cand(j) > lastc Then
                If bS = 0 Or Abs(cand(j) - tg) < Abs(bS - tg) Then bS = cand(j)
            End If
        Next j
        If bS = 0 Then bS = blen - 1
        If bS >= blen Then bS = blen - 1
        If bS <= lastc Then bS = lastc + 1
        cts(k) = bS: lastc = bS
    Next k
    cts(np) = blen
    cap2 = bn1 + np * (nCond + 2) + 16
    ReDim out2(1 To cap2): m2 = 0
    For i = 1 To bA
        L2_PushL out2, m2, cap2, base1(i)
    Next i
    For k = 1 To np
        pLen = cts(k) - cts(k - 1)
        If pLen > maxPart Then maxPart = pLen
        For i = bA + cts(k - 1) + 1 To bA + cts(k)
            L2_PushL out2, m2, cap2, base1(i)
        Next i
        If k < np Then
            ' копия end-строки БЕЗ kerr-хвоста (дублирование kerr недопустимо - Ф32 v21)
            pE2 = InStr(base1(be), ";")
            If pE2 > 0 Then
                L2_PushL out2, m2, cap2, Left(base1(be), pE2)
            Else
                L2_PushL out2, m2, cap2, base1(be)
            End If
            For i = ifStart To bA
                L2_PushL out2, m2, cap2, base1(i)
            Next i
        End If
    Next k
    For i = be To bn1
        L2_PushL out2, m2, cap2, base1(i)
    Next i
    ReDim Preserve base1(1 To m2)
    For i = 1 To m2
        base1(i) = out2(i)
    Next i
    bn1 = m2
    nParts = np
    msg = "Ф32а v21: блок " & CStr(blen) & " строк -> " & CStr(np) & " кусков, макс. кусок " & CStr(maxPart) & " (лимит в-блочной резки " & CStr(thrMono) & ")"
    L2_MonoSplit = True
End Function

' --- поиск строки декларации процедуры в interface (для вставки частей) ---
Private Function L2_FindDecl(ByRef lines() As String, ByVal impl As Long, ByVal nm As String) As Long
    Dim k As Long, t As String, dn As String
    L2_FindDecl = 0
    For k = 1 To impl - 1
        t = Trim(lines(k))
        If LCase(Left(t, 10)) = "procedure " Then
            dn = Mid(Trim(Mid(t, 11)), 1)
            If InStr(dn, ".") > 0 Then dn = Mid(dn, InStr(dn, ".") + 1)
            Dim sp1 As Long, sp2 As Long, sp3 As Long, dm As Long
            sp1 = InStr(dn, " "): sp2 = InStr(dn, "("): sp3 = InStr(dn, ";")
            dm = Len(dn) + 1
            If sp1 > 0 And sp1 < dm Then dm = sp1
            If sp2 > 0 And sp2 < dm Then dm = sp2
            If sp3 > 0 And sp3 < dm Then dm = sp3
            If dm > 1 Then dn = Trim(Left(dn, dm - 1)) Else dn = ""
            If StrComp(dn, nm, vbTextCompare) = 0 Then L2_FindDecl = k: Exit Function
        End If
    Next k
End Function


'=====================================================================
' ПОСТ-ПРОХОД: a/b-разбиение перегруженных процедур (LH2 v28)
' Порог = НАСТРАИВАЕМЫЙ лимит компилятора (v22; раньше - макс. исходника -
' при комплектах с исходными телами за лимитом разбивалось НОЛЬ процедур).
' Тело > порога делится на N частей (каждая <= порога). try-обёртка тела:
' кандидаты - границы узловых блоков ВНУТРИ try (dd=1). Переростки-блоки:
' тело блока выносится в части, вызовы - внутри блока. Локалы -> поля класса.
' Объявления частей - в interface. splitsD: имя -> число частей (для диспетчера).
'=====================================================================
Private Function L2_Split(ByVal fPath As String, ByVal thr As Long, ByRef nChunks As Long, ByRef splitsD As Object, ByRef mainOut As String) As Long
    Dim fn As Integer, lines() As String, n As Long, cap As Long, ln As String
    Dim i As Long, k As Long, t As String, dd As Long, kk As Long
    L2_Split = 0
    gNoParamSkipped = 0

    If Len(Dir(fPath)) = 0 Then Exit Function
    fn = FreeFile
    Open fPath For Input As #fn
    n = 0: cap = 1000: ReDim lines(1 To cap)
    Do While Not EOF(fn)
        Line Input #fn, ln
        n = n + 1
        If n > cap Then cap = cap * 2: ReDim Preserve lines(1 To cap)
        lines(n) = ln
    Loop
    Close #fn
    ReDim Preserve lines(1 To n)
    If n < 100 Then Exit Function
    If Len(mainOut) = 0 Then
        For i = 1 To n
            t = LCase(Trim(lines(i)))
            If Left(t, 10) = "procedure " And Right(t, 6) = ".work;" Then     ' v27: ".work;" = 6 символов; Right(t,7) никогда не совпадал -> mainOut не ставился, L2_Disp не вызывался
                mainOut = fPath: Exit For
            End If
        Next i
    End If
    Dim impl As Long: impl = 0
    For i = 1 To n
        If LCase(Trim(lines(i))) = "implementation" Then impl = i: Exit For
    Next i
    If impl = 0 Then Exit Function
    Dim cls As String: cls = L2_ClassName(lines, impl)
    ' --- фаза 1: инвентаризация процедур ---

Private Function L2_Disp(ByVal fPath As String, ByRef splitsD As Object) As Long
    Dim fn As Integer, lines() As String, n As Long, cap As Long, ln As String
    Dim i As Long, k As Long, t As String
    Dim out() As String, m As Long, ocap As Long
    Dim nb As Long: nb = 0
    L2_Disp = 0
    If Len(Dir(fPath)) = 0 Then Exit Function
    fn = FreeFile
    Open fPath For Input As #fn
    n = 0: cap = 1000: ReDim lines(1 To cap)
    Do While Not EOF(fn)
        Line Input #fn, ln
        n = n + 1
        If n > cap Then cap = cap * 2: ReDim Preserve lines(1 To cap)
        lines(n) = ln
    Loop
    Close #fn
    ReDim Preserve lines(1 To n)
    If n < 50 Then Exit Function
    ' предскан блоков вызовов: [gate/begin/Miss_comm/call/6-reg/end;/Log]
    Dim bS() As Long, bC() As Long, be() As Long, bL() As Long
    ReDim bS(1 To 64): ReDim bC(1 To 64): ReDim be(1 To 64): ReDim bL(1 To 64)
    i = 1
    Do While i <= n
        Dim matched As Boolean: matched = False
        t = Trim(lines(i))
        ' v26: было Left(LCase(t), 6) - 'work_1' по 6 символам = 'work_1' и диспетчер не срабатывал НИКОГДА
        If (Left(LCase(t), 5) = "work_" Or LCase(t) = "workzero;") And Right(t, 1) = ";" And InStr(t, " ") = 0 Then
            If i >= 4 Then
                Dim g As String: g = Trim(lines(i - 3))
                If InStr(LCase(g), "miss_ind[") > 0 And Trim(lines(i - 2)) = "begin" And InStr(lines(i - 1), "Miss_comm") > 0 Then
                    Dim e0 As Long: e0 = i
                    Do While e0 < n
                        If Trim(lines(e0 + 1)) = "end;" Then Exit Do
                        e0 = e0 + 1
                    Loop
                    nb = nb + 1
                    If nb > UBound(bS) Then
                        ReDim Preserve bS(1 To nb * 2): ReDim Preserve bC(1 To nb * 2)
                        ReDim Preserve be(1 To nb * 2): ReDim Preserve bL(1 To nb * 2)
                    End If
                    bS(nb) = i - 3: bC(nb) = i: be(nb) = e0 + 1: bL(nb) = e0 + 2
                    i = e0 + 2
                    matched = True
                End If
            End If
        End If
        If Not matched Then i = i + 1
    Loop
    Dim starts As Object: Set starts = CreateObject("Scripting.Dictionary")
    For i = 1 To nb: starts(CStr(bS(i))) = i: Next i
    ocap = n + 256: ReDim out(1 To ocap): m = 0
    i = 1
    Do While i <= n
        If starts.Exists(CStr(i)) Then
            Dim bi As Long: bi = starts(CStr(i))
            Dim ci As Long: ci = bC(bi)
            Dim nm As String: nm = Trim(Left(Trim(lines(ci)), Len(Trim(lines(ci))) - 1))
            Dim s As Long: s = 0
            For k = ci + 1 To ci + 6
                If k <= n Then
                    If InStr(lines(k), "Miss_proc[") > 0 Then
                        Dim sPos As Long: sPos = InStr(lines(k), "Miss_proc[")
                        Dim sEnd As Long: sEnd = InStr(sPos, lines(k), "]")
                        If sEnd > sPos Then s = CLng(Trim(Mid(lines(k), sPos + 11, sEnd - sPos - 11)))
                        Exit For
                    End If
                End If
            Next k
            If LCase(nm) = "workzero" Then
                L2_PushR out, m, ocap, lines, bS(bi), bL(bi)
            Else
                Dim nn As Long: nn = 0
                If LCase(Left(nm, 5)) = "work_" Then nn = CLng(Mid(nm, 6))
                If nn > 0 Then
                    Dim nPK As Long
                    nPK = 1
                    If splitsD.Exists(nm) Then nPK = CLng(splitsD(nm))
                    If nPK < 1 Then nPK = 1
                    Dim gate As String: gate = lines(bS(bi))
                    Dim lg As String
                    Dim pArr As Long, slK As Long, pnn As String
                    For pArr = 1 To nPK
                        If pArr = 1 Then
                            slK = 2 * nn - 1
                        ElseIf pArr = 2 Then
                            slK = 2 * nn
                        Else
                            If AB_SLOT > 99 Then slK = 99 Else slK = AB_SLOT: AB_SLOT = AB_SLOT + 1
                        End If
                        If nPK = 1 Then
                            pnn = nm
                        Else
                            pnn = nm & L2_Suf(pArr)
                        End If
                        If nPK > 1 Then L2_Disp = L2_Disp + 1
                        L2_PushL out, m, ocap, L2_SlotSub(gate, slK)
                        L2_PushL out, m, ocap, lines(bS(bi) + 1)
                        L2_PushL out, m, ocap, lines(bS(bi) + 2)
                        L2_PushL out, m, ocap, pnn & ";  // LH2 v22"
                        If nPK > 1 Then
                            L2_RegLinesDisp out, m, ocap, slK, pnn
                        Else
                            For k = ci + 1 To be(bi) - 1
                                L2_PushL out, m, ocap, L2_SlotAll(lines(k), slK)
                            Next k
                        End If
                        L2_PushL out, m, ocap, lines(be(bi))   ' v23: end; ВСЕГДА (раньше только при nPK>1 - терялся end; у нерезаных Work_N)
                        lg = lines(bL(bi))
                        If nPK > 1 Then lg = Replace(lg, " " & nm & " - SUCSESS ", " " & pnn & " - SUCSESS ")
                        L2_PushL out, m, ocap, lg
                    Next pArr
                Else
                    ' не Work_N (WorkZero и прочее) - без изменений
                    L2_PushR out, m, ocap, lines, bS(bi), bL(bi)
                End If
            End If
            i = bL(bi) + 1
        Else
            L2_PushL out, m, ocap, lines(i)
            i = i + 1
        End If
    Loop
    fn = FreeFile
    Open fPath For Output As #fn
    For i = 1 To m
        Print #fn, out(i)
    Next i
    Close #fn
End Function

' Ф55: СТОП-инвариант приёмки. Меряет ТЕЛО процедуры dd-сканом от begin до парного end.
' Вызывается ПОСЛЕ последнего прохода резки по выходному файлу; накапливает нарушения.
' Решение о выпуске комплекта принимает основной процесс по LH3_StopFlag.
Private Sub LH3_StopCheck(ByVal pth As String, ByVal outn As String)
    Dim fPath As String, f1 As Integer
    Dim L() As String, n As Long, ln As String
    Dim i As Long, k As Long, hdr As Long, bi As Long, ei As Long
    Dim dd As Long, body As Long, t As String, nm As String
    fPath = pth & outn
    If Dir(fPath) = "" Then Exit Sub
    ReDim L(1 To 4096): n = 0
    f1 = FreeFile
    Open fPath For Input As #f1
    Do While Not EOF(f1)
        If n = UBound(L) Then ReDim Preserve L(1 To UBound(L) * 2)
        n = n + 1
        Line Input #f1, ln
        L(n) = ln
    Loop
    Close #f1
    If n = 0 Then Exit Sub
    ReDim Preserve L(1 To n)
    i = 1
    Do While i <= n
        t = Trim(L(i))
        If Left(t, 10) = "procedure " Then
            hdr = i
            Do While hdr <= n
                If Right(Trim(L(hdr)), 1) = ";" Then Exit Do
                hdr = hdr + 1
            Loop
            If hdr > n Then Exit Do
            nm = Trim(L(hdr))
            nm = Left(nm, Len(nm) - 1)
            If InStr(nm, "(") > 0 Then nm = Trim(Left(nm, InStr(nm, "(") - 1))
            bi = 0
            For k = hdr + 1 To n
                t = Trim(L(k))
                If t = "begin" Then
                    bi = k
                    Exit For
                End If
                If Left(t, 10) = "procedure " Or Left(t, 9) = "function " Then Exit For
                If t = "implementation" Or t = "end." Then Exit For
            Next k
            If bi > 0 Then
                dd = 0: ei = 0
                For k = bi + 1 To n
                    dd = dd + L2_Delta(L(k))
                    If dd < 0 Then
                        ei = k
                        Exit For
                    End If
                Next k
                If ei > bi Then
                    body = ei - bi - 1
                    If body > LH3_SAFE_LINE Then
                        LH3_StopCnt = LH3_StopCnt + 1
                        LH3_StopFlag = True
                        If LH3_StopCnt <= 10 Then
                            LH3_StopMsg = LH3_StopMsg & "  " & nm & " = " & CStr(body) & " строк  (" & outn & ")" & vbCrLf
                        End If
                    End If
                    i = ei
                End If
            End If
        End If
        i = i + 1
    Loop
End Sub

' --- регулярная замена (VBScript.RegExp, v4) ---
Private Function L4_Re(ByVal s As String, ByVal pat As String, ByVal rep As String, ByVal nCnt As Long) As String
    Static re As Object, mc As Object
    L4_Re = s
    If re Is Nothing Then Set re = CreateObject("VBScript.RegExp")   ' v4: объект кэшируется - иначе CreateObject на каждый вызов
    If re Is Nothing Then
        L1_Log "L4_Re ERROR: объект RegExp недоступен pat=" & Left(pat, 120)   ' LH3 v19 Ф30
        Exit Function
    End If
    On Error Resume Next
    Err.Clear
    re.Global = True: re.MultiLine = True: re.IgnoreCase = False
    re.Pattern = pat
    Set mc = re.Execute(s)
    On Error GoTo 0
    If Err.Number <> 0 Or mc Is Nothing Then
        ' LH3 v19 Ф30: паттерн не принят - лог и возврат исходной строки
        L1_Log "L4_Re ERROR err=" & CStr(Err.Number) & " " & Err.Description & " pat=" & Left(pat, 120)
        Err.Clear
        Exit Function
    End If
    nCnt = mc.Count
    If nCnt > 0 Then L4_Re = re.Replace(s, rep) Else L4_Re = s
End Function

' --- проверка совпадения и подстрока по группе ---
Private Function L4_M(ByVal s As String, ByVal pat As String) As Boolean
    Static re As Object
    L4_M = False
    If re Is Nothing Then Set re = CreateObject("VBScript.RegExp")   ' v4: объект кэшируется
    If re Is Nothing Then
        L1_Log "L4_M ERROR: объект RegExp недоступен pat=" & Left(pat, 120)   ' LH3 v19 Ф30
        Exit Function
    End If
    On Error Resume Next
    Err.Clear
    re.Global = False: re.IgnoreCase = False
    re.Pattern = pat
    Dim bM As Boolean
    bM = re.Test(s)
    On Error GoTo 0
    If Err.Number <> 0 Then
        ' LH3 v19 Ф30: паттерн не принят - лог и деградация к False
        L1_Log "L4_M ERROR err=" & CStr(Err.Number) & " " & Err.Description & " pat=" & Left(pat, 120)
        Err.Clear
        Exit Function
    End If
    L4_M = bM
End Function
Private Function L4_SM(ByVal s As String, ByVal pat As String, ByVal idx As Long) As String
    Static re As Object, mc As Object
    If re Is Nothing Then Set re = CreateObject("VBScript.RegExp")   ' v4: объект кэшируется
    re.Global = False: re.IgnoreCase = False
    On Error Resume Next
    Err.Clear
    re.Pattern = pat
    Set mc = re.Execute(s)
    On Error GoTo 0
    If Err.Number <> 0 Or mc Is Nothing Then
        ' LH3 v18 Ф27: паттерн (или объект) не принят - фиксируем в лог и деградируем к ""
        L1_Log "L4_SM ERROR err=" & CStr(Err.Number) & " " & Err.Description & " pat=" & Left(pat, 120)
        Err.Clear
        L4_SM = "": Exit Function
    End If
    If mc.Count > 0 Then
        If mc(0).SubMatches.Count > idx Then
            L4_SM = mc(0).SubMatches(idx)
        Else
            L4_SM = ""   ' LH3 v12: группы нет - пусто вместо Err 5
        End If
    Else
        L4_SM = ""
    End If
End Function
