Option Explicit
Private csh1 As Boolean

' --- простой идентификатор (безопасно гасить присваиванием) ---
Private Function L1_Simple(ByVal s As String) As Boolean
    Dim i As Long, c As String
    L1_Simple = False
    If Len(s) = 0 Then Exit Function
    For i = 1 To Len(s)
        c = Mid(s, i, 1)
        If Not ((c >= "a" And c <= "z") Or (c >= "A" And c <= "Z") Or (c >= "0" And c <= "9") Or c = "_") Then Exit Function
    Next i
    L1_Simple = True
End Function
' --- дельта глубины begin/try/repeat/case - end/until (вне строк-литералов) ---
Private Function L1_Delta(ByVal s As String) As Long
    Dim q As Boolean, i As Long, c As String, buf As String, t As String
    Dim d As Long, j As Long, w As String
    q = False: buf = ""
    For i = 1 To Len(s)
        c = Mid(s, i, 1)
        If c = "'" Then
            q = Not q: buf = buf & c
        ElseIf c = "/" And (Not q) And i < Len(s) And Mid(s, i + 1, 1) = "/" Then
            Exit For
        Else
            buf = buf & c
        End If
    Next i
    t = LCase(buf)
    If InStr(t, "begin") = 0 And InStr(t, "end") = 0 And InStr(t, "try") = 0 _
       And InStr(t, "repeat") = 0 And InStr(t, "until") = 0 And InStr(t, "case") = 0 Then
        L1_Delta = 0: Exit Function
    End If
    For i = 1 To Len(t)
        c = Mid(t, i, 1)
        If Not ((c >= "a" And c <= "z") Or (c >= "0" And c <= "9") Or c = "_") Then Mid(t, i, 1) = " "
    Next i
    d = 0
    Dim parts() As String
    parts = Split(t, " ")
    For j = LBound(parts) To UBound(parts)
        w = parts(j)
        If w = "begin" Or w = "try" Or w = "repeat" Then d = d + 1
        If w = "end" Or w = "until" Then d = d - 1
        If w = "case" Then d = d + 1
    Next j
    L1_Delta = d
End Function


' --- есть ли кириллица ВНЕ строк-литералов и вне "//" (Ф4) ---
Private Function L1_CyrOut(ByVal s As String) As Boolean
    Dim q As String, i As Long, c As String, a As Long
    q = ""
    For i = 1 To Len(s)
        c = Mid(s, i, 1)
        If q <> "" Then
            If c = q Then q = ""
        ElseIf c = "'" Or c = Chr(34) Then
            q = c
        ElseIf c = "/" And i < Len(s) And Mid(s, i + 1, 1) = "/" Then
            Exit For
        Else
            a = Asc(c)
            If (a >= 192 And a <= 255) Or a = 168 Or a = 184 Then
                L1_CyrOut = True: Exit Function
            End If
        End If
    Next i
    L1_CyrOut = False
End Function

' --- условие NaN/бред ---
Private Function L1_Bad(ByVal nm As String) As String
    If csh1 Then
        L1_Bad = "double.IsNaN(" & nm & ")||Math.Abs(" & nm & ")>1e300"
    Else
        L1_Bad = "IsNan(" & nm & ") or (Abs(" & nm & ")>1e300)"
    End If
End Function

' --- запись события (первое появление = порядковый номер) ---
Private Function L1_Rec(ByVal idx As String, ByVal nm As String, ByVal uz As Long, ByVal tp As Long) As String
    If csh1 Then
        L1_Rec = "if (" & idx & "==0) && (NaN_seq<1000) { NaN_seq++; " & idx & "=NaN_seq; NaN_nm[NaN_seq]=""" & nm & """; NaN_uz[NaN_seq]=" & CStr(uz) & "; NaN_tp[NaN_seq]=" & CStr(tp) & "; }"
    Else
        L1_Rec = "if (" & idx & "=0) and (NaN_seq<1000) then begin NaN_seq:=NaN_seq+1; " & idx & ":=NaN_seq; NaN_nm[NaN_seq]:='" & nm & "'; NaN_uz[NaN_seq]:=" & CStr(uz) & "; NaN_tp[NaN_seq]:=" & CStr(tp) & "; end;"
    End If
End Function

' --- условие вне диапазона давлений (активен при LIM_P_max>0, инициализация в _0i) ---
