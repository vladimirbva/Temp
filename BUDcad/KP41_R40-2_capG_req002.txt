                    double tol2 = Math.Max(1e-3, Math.Max(1e-4 * Math.Max(inf2, out2), 0.01));
                    if (Math.Abs(inf2 - out2) > tol2) perNodeOk = false;
                }
                res.NodeBalanceOk = allConverged || perNodeOk;
                // К2: MaxImbalanceRel учитывается в вердикте — фактический баланс в допуске,
                // даже если решатель не дожал ε (мёртвая зона застоя) — сходимость признаётся.
                if (perNodeOk && !allConverged) res.Converged = true;
            }
            // v85 (КП2.1-В8): честная пометка достоверности Psat + гейт-провалы констант в Warnings.
            res.PsatConfidenceNote = PsatConfidenceNote(objects);
            foreach (var gv in ComponentLibrary.PsatGateViolations)
                if (!res.Warnings.Contains(gv)) res.Warnings.Add(gv);

            foreach (var e in res.Edges)
            {
                var a = (e.IndexA >= 0) ? res.Nodes[e.IndexA] : res.FindNode(e.NodeAKey);
                var b = (e.IndexB >= 0) ? res.Nodes[e.IndexB] : res.FindNode(e.NodeBKey);
                if (a != null && b != null && !double.IsNaN(a.Pressure) && !double.IsNaN(b.Pressure)) e.Dp = a.Pressure - b.Pressure;
                double v = Math.Abs(e.Q) * 4.0 / (Math.PI * Math.Pow(Math.Max(1e-9, e.Diameter), 2));
                e.Mode = Math.Abs(e.Q) < 1e-9 ? "застой" : (v < 0.03 ? "ламинарный" : "турбулентный");
            }

            // ─── v2.3-w18: пост-обработка 127 ч.2б + п.124 (до Propagate, чтобы выноски видели клампы) ───
            try
            {
                foreach (var fo in byId.Values.Where(x => x != null && x.Kind == ObjectKind.Furnace))
                {
                    var fuelNode18 = res.Nodes.FirstOrDefault(n => n.ObjId == fo.Id && n.Role == NozzleRole.FurnaceFuel);
                    var airNode18 = res.Nodes.FirstOrDefault(n => n.ObjId == fo.Id && n.Role == NozzleRole.FurnaceAir);
                    if (fuelNode18 == null || airNode18 == null) continue;
                    double qFuel18 = res.Edges.Where(e => e.NodeBKey == fuelNode18.Key).Sum(e => Math.Max(0.0, e.Q)); // м³/с
                    if (qFuel18 <= 1e-12) continue;
                    var moleF18 = fuelNode18.MoleFractions;
                    if (moleF18 == null) continue;
                    double o2F18 = FurnaceEngine.O2MolPerMol(moleF18);
                    double airPerFuel18 = (1.0 / 0.21) * o2F18; // моль возд./моль топлива (объёмные доли те же)
                    double lambda18 = FurnaceEngine.AIR_EXCESS_LAMBDA;
                    double qAirMax18 = qFuel18 * airPerFuel18 * lambda18;
                    var airEdges18 = res.Edges.Where(e => e.NodeBKey == airNode18.Key).ToList();
                    double qAirNow18 = airEdges18.Sum(e => Math.Max(0.0, e.Q));
                    if (qAirNow18 > qAirMax18 * 1.0001 && qAirNow18 > 1e-12)
                    {
                        double scale18 = qAirMax18 / qAirNow18;
                        foreach (var ae in airEdges18) if (ae.Q > 0) ae.Q *= scale18;
                        res.Warnings.Add($"Печь «{fo.Name}»: расход воздуха ограничен потребностью сгорания (λ={lambda18:0.##}): было {qAirNow18 * 3600:0.#} м³/ч, стало {qAirMax18 * 3600:0.#} м³/ч (стехиометрия состава топлива).");
                    }
                }
            }
            catch (Exception exA18) { res.Warnings.Add("Кламп воздуха печи: " + exA18.Message); }

            try
            {
                // п.124 (демпфер): при застое насосного ребра (|Q|≈0) всас «отсасывает» невязку
                // (у Димы P вх = 1,05 при источнике 1,0 — обратный ток через ребро насоса R=1e3).
                // Выравнивание: если ВСЕ внешние рёбра узла всаса застойные и среди соседей есть
                // граница — P всаса := макс давление граничных соседей (статическое выравнивание).
                foreach (var pe18 in res.Edges.Where(e => e.IsPump && Math.Abs(e.Q) < 1e-9))
                {
                    var su18 = res.FindNode(pe18.NodeAKey);
                    if (su18 == null || su18.IsBoundary) continue;
                    bool hasLive = false; double pMax18 = double.NaN;
                    foreach (var se18 in res.Edges)
                    {
                        if (ReferenceEquals(se18, pe18) || se18.IsPump) continue;
                        string otherK18 = se18.NodeAKey == su18.Key ? se18.NodeBKey : (se18.NodeBKey == su18.Key ? se18.NodeAKey : null);
                        if (otherK18 == null) continue;
                        if (Math.Abs(se18.Q) >= 1e-9) { hasLive = true; break; }
                        var other18 = res.FindNode(otherK18);
                        if (other18 != null && other18.IsBoundary)
                            pMax18 = double.IsNaN(pMax18) ? other18.Pressure : Math.Max(pMax18, other18.Pressure);
                    }
                    if (!hasLive && !double.IsNaN(pMax18)) su18.Pressure = pMax18;
                }
            }
            catch (Exception exD18) { res.Warnings.Add("Демпфер застойного насоса: " + exD18.Message); }

            TemperaturePropagator.Propagate(res, adjacency, settings, byId, settings.SimMode ? 10 : 200, state);

            for (int i = 0; i < spec.Edges.Count; i++)
            { var he = res.Edges[i]; var a = (he.IndexA >= 0) ? res.Nodes[he.IndexA] : null; if (a != null && a.MoleFractions != null) dyn.Density[i] = CalculateDensity(a.MoleFractions, a.Temperature, a.Pressure, settings.UsePR); } // v65+v70: живая Density в dyn, PR — из снапшота настроек

            // v55.13 (реестр п.24): отчёт струи эжектора (первый тик / смена режима)
            ReportEjectorJets(res, objects);
            return res;
        }
