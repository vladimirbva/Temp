        private static (List<PinchStream> streams, double hoursPerYear, List<string> warnings) ParseXlsxCore(byte[] data, bool readMapping, out PinchMappingData mapping)
        {
            var streams = new List<PinchStream>();
            var warnings = new List<string>();
            double hours = 8000;
            bool hoursParsed = false; // v37 (ревью 19): реальный ввод «8000» не должен считаться «не распарсенным»
            mapping = null;

            using (var ms = new MemoryStream(data))
            using (var zip = new ZipArchive(ms, ZipArchiveMode.Read))
            {
                var shared = ReadShared(zip);
                var sheets = SheetFiles(zip);   // Сп2-эт3: порядок и имена листов книги
                if (sheets.Count == 0)
                {
                    // прежний fallback: sheet1.xml, иначе первый xl/worksheets/sheet*.xml
                    var legacy = zip.Entries.FirstOrDefault(e => e.FullName == "xl/worksheets/sheet1.xml")
                              ?? zip.Entries.FirstOrDefault(e => e.FullName.StartsWith("xl/worksheets/sheet", StringComparison.OrdinalIgnoreCase) && e.FullName.EndsWith(".xml"));
                    if (legacy != null) sheets.Add((null, legacy.FullName));
                }

                ZipArchiveEntry sheetEntry = null;
                List<Dictionary<string, string>> rows = null;
                var named = sheets.FirstOrDefault(s => s.Name == "Потоки");
                if (named.Path != null)
                {
                    sheetEntry = zip.GetEntry(named.Path);
                }
                else
                {
                    // чужой файл: первый лист с шапкой «Название» (прежний смысл «первого листа»)
                    foreach (var s in sheets)
                    {
                        var e2 = zip.GetEntry(s.Path);
                        if (e2 == null) continue;
                        var r2 = ReadSheetRows(e2, shared);
                        if (r2 != null && r2.Any(d => d.Values.Any(v => v != null && v.Trim().Equals("Название", StringComparison.OrdinalIgnoreCase))))
                        {
                            sheetEntry = e2;
                            rows = r2;
                            break;
                        }
                    }
                    if (sheetEntry == null && sheets.Count > 0) sheetEntry = zip.GetEntry(sheets[0].Path);
                }
                if (sheetEntry == null) { warnings.Add("В файле нет листов."); return (streams, hours, warnings); }
                if (rows == null) rows = ReadSheetRows(sheetEntry, shared);
                if (rows == null) { warnings.Add("Лист «Потоки» пуст."); return (streams, hours, warnings); }

                // Поиск шапки: строка с ячейкой «Название»
                int headRow = -1;
                Dictionary<string, string> head = null;
                for (int i = 0; i < rows.Count; i++)
                {
                    foreach (var kv in rows[i])
                        if (kv.Value != null && kv.Value.Trim().Equals("Название", StringComparison.OrdinalIgnoreCase)) { headRow = i; head = rows[i]; break; }
                    if (headRow >= 0) break;
                }
                if (headRow < 0) { warnings.Add("Шапка «Название» не найдена — это не клешё BUDcad."); return (streams, hours, warnings); }

                var cols = new Dictionary<string, string>();
                foreach (var kv in head)
                {
                    string v = (kv.Value ?? "").Trim();
                    if (v.Length == 0) continue;
                    cols[v] = ColOf(kv.Key);
                }
                string GetCol(string headerName)
                {
                    foreach (var kv in cols)
                        if (kv.Key.StartsWith(headerName, StringComparison.OrdinalIgnoreCase)) return kv.Value;
                    return null;
                }
                string colName = GetCol("Название"), colType = GetCol("Тип"), colTs = GetCol("T нач"),
                       colTt = GetCol("T кон"), colCp = GetCol("m·cp"), colUtil = GetCol("Утилита"),
                       colPrice = GetCol("Цена"), colFact = GetCol("Факт"), colNote = GetCol("Примечание");

                for (int i = headRow + 1; i < rows.Count; i++)
                {
                    var row = rows[i];
                    string name = colName != null && row.TryGetValue(colName, out var n) ? n : null;
                    string all = string.Join(" ", row.Values.Where(v => v != null));
                    if (all.Contains("Часов работы в год"))
                    {
                        foreach (var kv in row)
                        {
                            double h;
                            if (double.TryParse(kv.Value, NumberStyles.Any, CultureInfo.InvariantCulture, out h) && h > 0 && !kv.Value.TrimStart().StartsWith("Часов")) // v38 (ревью 6): проверка ЯЧЕЙКИ (all = вся строка → условие всегда ложно)
                            { hours = h; hoursParsed = true; break; }
                        }
                        // число часов — в СЛЕДУЮЩЕЙ колонке строки «Часов работы в год:»
                        if (!hoursParsed)
                        {
                            string key = null;
                            foreach (var kv in row) if (kv.Value != null && kv.Value.Contains("Часов")) { key = kv.Key; break; }
                            if (key != null)
                            {
                                int colIdx = ColIndexOf(key);
                                foreach (var kv in row)
                                    if (kv.Key.StartsWith(ColName(colIdx + 1)))
                                        if (double.TryParse(kv.Value, NumberStyles.Any, CultureInfo.InvariantCulture, out hours) || double.TryParse(kv.Value, NumberStyles.Any, CultureInfo.CurrentCulture, out hours)) break;
                            }
                        }
                        continue;
                    }
                    if (all.Contains("СПРАВКА") || all.Contains("ЭСП (энергосберегающий")) break;
                    if (string.IsNullOrWhiteSpace(name)) continue;
                    if (name.Trim().Equals("№", StringComparison.OrdinalIgnoreCase)) continue;

                    string type = colType != null && row.TryGetValue(colType, out var t) ? t : "";
                    bool hot = type.Contains("Гор");
                    double Ts = colTs != null && row.TryGetValue(colTs, out var a) ? ParseD(a) : 0;
                    double Tt = colTt != null && row.TryGetValue(colTt, out var b) ? ParseD(b) : 0;
                    double cp = colCp != null && row.TryGetValue(colCp, out var c) ? ParseD(c) : 0;
                    var st = new PinchStream { Name = name.Trim(), IsHot = hot, Ts = Ts, Tt = Tt, CpKW = cp };
                    if (colUtil != null && row.TryGetValue(colUtil, out var u)) st.UtilityName = u.Trim();
                    if (colPrice != null && row.TryGetValue(colPrice, out var pr)) { double d = ParseD(pr); st.UtilityCostRubKwh = d > 0 ? d : double.NaN; }
                    if (colFact != null && row.TryGetValue(colFact, out var f)) { double d = ParseD(f); st.UtilityFactKW = d > 0 ? d : double.NaN; }
                    if (colNote != null && row.TryGetValue(colNote, out var nt)) st.Note = nt.Trim();
                    if (st.CpKW <= 0) warnings.Add($"«{st.Name}»: m·cp не задан или ≤ 0 — поток будет пропущен при расчёте.");
                    streams.Add(st);
                }
                if (readMapping) mapping = ReadMappingSheet(zip, sheets, shared, warnings);
            }
            return (streams, hours, warnings);
        }
